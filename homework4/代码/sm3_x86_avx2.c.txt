#include "sm3.h"
#include <immintrin.h>

#define ROTL32(x, n) (((x) << (n)) | ((x) >> (32 - (n))))
#define P0(x)        ((x) ^ ROTL32(x, 9)  ^ ROTL32(x, 17))
#define P1(x)        ((x) ^ ROTL32(x, 15) ^ ROTL32(x, 23))

#define FF0(x,y,z)   ((x) ^ (y) ^ (z))
#define GG0(x,y,z)   ((x) ^ (y) ^ (z))
#define FF1(x,y,z)   (((x) & (y)) | ((x) & (z)) | ((y) & (z)))
#define GG1(x,y,z)   (((x) & (y)) | ((~(x)) & (z)))

#define T_J0 0x79cc4519UL
#define T_J1 0x7a879d8aUL

static inline __m128i _mm_rotl_epi32(__m128i x, int n)
{
    return _mm_or_si128(_mm_slli_epi32(x, n), _mm_srli_epi32(x, 32 - n));
}

static inline __m128i mm_P1_epi32(__m128i x)
{
    __m128i x15 = _mm_rotl_epi32(x, 15);
    __m128i x23 = _mm_rotl_epi32(x, 23);
    return _mm_xor_si128(_mm_xor_si128(x, x15), x23);
}

static inline __m128i sm3_avx2_expand4(
    __m128i w0_3, __m128i w3_6, __m128i w6_9,
    __m128i w9_12, __m128i w12_15)
{
    __m128i wj3_rot15 = _mm_rotl_epi32(w9_12, 15);
    __m128i tmp = _mm_xor_si128(w0_3, w6_9);
    tmp = _mm_xor_si128(tmp, wj3_rot15);
    tmp = mm_P1_epi32(tmp);

    __m128i wj13_rot7 = _mm_rotl_epi32(w3_6, 7);
    __m128i res = _mm_xor_si128(tmp, wj13_rot7);
    res = _mm_xor_si128(res, w6_9);
    return res;
}

#define SM3_ROUND(A,B,C,D,E,F,G,H, wj, wpj, tj, idx) \
do { \
    uint32_t a12 = ROTL32(A, 12); \
    uint32_t ss1 = ROTL32(a12 + E + ROTL32(tj, idx), 7); \
    uint32_t ss2 = ss1 ^ a12; \
    uint32_t ff = (idx < 16) ? FF0(A,B,C) : FF1(A,B,C); \
    uint32_t gg = (idx < 16) ? GG0(E,F,G) : GG1(E,F,G); \
    uint32_t tt1 = ff + D + ss2 + wpj; \
    uint32_t tt2 = gg + H + ss1 + wj; \
    D = C; C = ROTL32(B,9); B = A; A = tt1; \
    H = G; G = ROTL32(F,19); F = E; E = P0(tt2); \
} while(0)

#define SM3_4GROUP(A,B,C,D,E,F,G,H, wbuf, wpbuf, tj, start) \
do { \
    SM3_ROUND(A,B,C,D,E,F,G,H, wbuf[0], wpbuf[0], tj, start+0); \
    SM3_ROUND(D,A,B,C,H,E,F,G, wbuf[1], wpbuf[1], tj, start+1); \
    SM3_ROUND(C,D,A,B,G,H,E,F, wbuf[2], wpbuf[2], tj, start+2); \
    SM3_ROUND(B,C,D,A,F,G,H,E, wbuf[3], wpbuf[3], tj, start+3); \
} while(0)

void sm3_compress(uint32_t state[8], const uint8_t block[64])
{
    __m128i bswap_mask = _mm_set_epi8(12,13,14,15,8,9,10,11,4,5,6,7,0,1,2,3);
    __m128i w0_3   = _mm_shuffle_epi8(_mm_loadu_si128((const __m128i*)(block+0)), bswap_mask);
    __m128i w4_7   = _mm_shuffle_epi8(_mm_loadu_si128((const __m128i*)(block+16)), bswap_mask);
    __m128i w8_11  = _mm_shuffle_epi8(_mm_loadu_si128((const __m128i*)(block+32)), bswap_mask);
    __m128i w12_15 = _mm_shuffle_epi8(_mm_loadu_si128((const __m128i*)(block+48)), bswap_mask);

    __m128i w3_6  = _mm_alignr_epi8(w4_7, w0_3, 12);
    __m128i w6_9  = _mm_alignr_epi8(w8_11, w4_7, 12);
    __m128i w9_12 = _mm_alignr_epi8(w12_15, w8_11, 12);

    uint32_t A=state[0], B=state[1], C=state[2], D=state[3];
    uint32_t E=state[4], F=state[5], G=state[6], H=state[7];

    uint32_t w_buf[4], wp_buf[4];

    for (int i = 0; i < 4; i++)
    {
        __m128i w_cur, w_next;
        switch(i){
            case 0:w_cur=w0_3;w_next=w4_7;break;
            case 1:w_cur=w4_7;w_next=w8_11;break;
            case 2:w_cur=w8_11;w_next=w12_15;break;
            default:w_cur=w12_15;w_next=_mm_setzero_si128();break;
        }
        _mm_storeu_si128((__m128i*)w_buf, w_cur);
        _mm_storeu_si128((__m128i*)wp_buf, _mm_xor_si128(w_cur, w_next));

        SM3_4GROUP(A,B,C,D,E,F,G,H, w_buf, wp_buf, T_J0, i*4);

        if(i<3){
            __m128i new_w = sm3_avx2_expand4(w0_3,w3_6,w6_9,w9_12,w12_15);
            w0_3=w4_7;w4_7=w8_11;w8_11=w12_15;w12_15=new_w;
            w3_6=_mm_alignr_epi8(w4_7,w0_3,12);
            w6_9=_mm_alignr_epi8(w8_11,w4_7,12);
            w9_12=_mm_alignr_epi8(w12_15,w8_11,12);
        }
    }

    for (int i = 4; i < 16; i++)
    {
        _mm_storeu_si128((__m128i*)w_buf, w0_3);
        _mm_storeu_si128((__m128i*)wp_buf, _mm_xor_si128(w0_3, w4_7));
        SM3_4GROUP(A,B,C,D,E,F,G,H, w_buf, wp_buf, T_J1, i*4);

        __m128i new_w = sm3_avx2_expand4(w0_3,w3_6,w6_9,w9_12,w12_15);
        w0_3=w4_7;w4_7=w8_11;w8_11=w12_15;w12_15=new_w;
        w3_6=_mm_alignr_epi8(w4_7,w0_3,12);
        w6_9=_mm_alignr_epi8(w8_11,w4_7,12);
        w9_12=_mm_alignr_epi8(w12_15,w8_11,12);
    }

    state[0]^=A;state[1]^=B;state[2]^=C;state[3]^=D;
    state[4]^=E;state[5]^=F;state[6]^=G;state[7]^=H;
}
